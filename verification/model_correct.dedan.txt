// =========================================================================
// CORRECT MODEL: Producer-Consumer with semaphore-based synchronization
// Lab 3 formal verification — server view
// =========================================================================
// System: 3 producers (PA, PB, PC), 2 consumers (C1, C2)
// Shared resources: mutex (binary), slots (counting=4), wake_c1/wake_c2 (binary)
// Queue states: q_abc where a,b,c in {0,1} — presence of type A/B/C elements
// Abstraction: each producer generates exactly 1 element, capacity = 4
// =========================================================================

system ProdConsLab3;

// --- Binary semaphore: mutex ---
// Protects the shared queue (critical section).
// All 5 agents (PA, PB, PC, C1, C2) can wait/signal.
server:	mutex (servers SPA:SPA, SPB:SPB, SPC:SPC, SC1:SC1, SC2:SC2; agents PA:PA, PB:PB, PC:PC, C1:C1, C2:C2),
services	{wait, signal},
states	{up, down},
actions {
	{PA.mutex.wait, mutex.up}	-> {PA.SPA.ok_mutex, mutex.down},
	{PB.mutex.wait, mutex.up}	-> {PB.SPB.ok_mutex, mutex.down},
	{PC.mutex.wait, mutex.up}	-> {PC.SPC.ok_mutex, mutex.down},
	{C1.mutex.wait, mutex.up}	-> {C1.SC1.ok_mutex, mutex.down},
	{C2.mutex.wait, mutex.up}	-> {C2.SC2.ok_mutex, mutex.down},
	{PA.mutex.signal, mutex.down}	-> {PA.SPA.ok_rel, mutex.up},
	{PA.mutex.signal, mutex.up}	-> {PA.SPA.ok_rel, mutex.up},
	{PB.mutex.signal, mutex.down}	-> {PB.SPB.ok_rel, mutex.up},
	{PB.mutex.signal, mutex.up}	-> {PB.SPB.ok_rel, mutex.up},
	{PC.mutex.signal, mutex.down}	-> {PC.SPC.ok_rel, mutex.up},
	{PC.mutex.signal, mutex.up}	-> {PC.SPC.ok_rel, mutex.up},
	{C1.mutex.signal, mutex.down}	-> {C1.SC1.ok_rel, mutex.up},
	{C1.mutex.signal, mutex.up}	-> {C1.SC1.ok_rel, mutex.up},
	{C2.mutex.signal, mutex.down}	-> {C2.SC2.ok_rel, mutex.up},
	{C2.mutex.signal, mutex.up}	-> {C2.SC2.ok_rel, mutex.up},
};

// --- Counting semaphore: slots (0..4) ---
// Blocks producers when queue is full (all 4 slots occupied).
// Producers do wait (decrement), consumers do signal (increment).
server:	slots (servers SPA:SPA, SPB:SPB, SPC:SPC, SC1:SC1, SC2:SC2; agents PA:PA, PB:PB, PC:PC, C1:C1, C2:C2),
services	{wait, signal},
states	{s0, s1, s2, s3, s4},
actions {
	{PA.slots.wait, slots.s4}	-> {PA.SPA.ok_slots, slots.s3},
	{PA.slots.wait, slots.s3}	-> {PA.SPA.ok_slots, slots.s2},
	{PA.slots.wait, slots.s2}	-> {PA.SPA.ok_slots, slots.s1},
	{PA.slots.wait, slots.s1}	-> {PA.SPA.ok_slots, slots.s0},
	{PB.slots.wait, slots.s4}	-> {PB.SPB.ok_slots, slots.s3},
	{PB.slots.wait, slots.s3}	-> {PB.SPB.ok_slots, slots.s2},
	{PB.slots.wait, slots.s2}	-> {PB.SPB.ok_slots, slots.s1},
	{PB.slots.wait, slots.s1}	-> {PB.SPB.ok_slots, slots.s0},
	{PC.slots.wait, slots.s4}	-> {PC.SPC.ok_slots, slots.s3},
	{PC.slots.wait, slots.s3}	-> {PC.SPC.ok_slots, slots.s2},
	{PC.slots.wait, slots.s2}	-> {PC.SPC.ok_slots, slots.s1},
	{PC.slots.wait, slots.s1}	-> {PC.SPC.ok_slots, slots.s0},
	{C1.slots.signal, slots.s0}	-> {C1.SC1.ok_slots, slots.s1},
	{C1.slots.signal, slots.s1}	-> {C1.SC1.ok_slots, slots.s2},
	{C1.slots.signal, slots.s2}	-> {C1.SC1.ok_slots, slots.s3},
	{C1.slots.signal, slots.s3}	-> {C1.SC1.ok_slots, slots.s4},
	{C2.slots.signal, slots.s0}	-> {C2.SC2.ok_slots, slots.s1},
	{C2.slots.signal, slots.s1}	-> {C2.SC2.ok_slots, slots.s2},
	{C2.slots.signal, slots.s2}	-> {C2.SC2.ok_slots, slots.s3},
	{C2.slots.signal, slots.s3}	-> {C2.SC2.ok_slots, slots.s4},
};

// --- Binary semaphore: wake_c1 ---
// Wakes Consumer C1. Signaled by producers PA and PB after pushing A or B.
// Only C1 waits on it.
server:	wake_c1 (servers SPA:SPA, SPB:SPB, SC1:SC1; agents PA:PA, PB:PB, C1:C1),
services	{wait, signal},
states	{up, down},
actions {
	{C1.wake_c1.wait, wake_c1.up}		-> {C1.SC1.ok_wc1, wake_c1.down},
	{PA.wake_c1.signal, wake_c1.down}	-> {PA.SPA.ok_sig, wake_c1.up},
	{PA.wake_c1.signal, wake_c1.up}	-> {PA.SPA.ok_sig, wake_c1.up},
	{PB.wake_c1.signal, wake_c1.down}	-> {PB.SPB.ok_sig, wake_c1.up},
	{PB.wake_c1.signal, wake_c1.up}	-> {PB.SPB.ok_sig, wake_c1.up},
};

// --- Binary semaphore: wake_c2 ---
// Wakes Consumer C2. Signaled by producer PC after pushing C.
// Only C2 waits on it.
server:	wake_c2 (servers SPC:SPC, SC2:SC2; agents PC:PC, C2:C2),
services	{wait, signal},
states	{up, down},
actions {
	{C2.wake_c2.wait, wake_c2.up}		-> {C2.SC2.ok_wc2, wake_c2.down},
	{PC.wake_c2.signal, wake_c2.down}	-> {PC.SPC.ok_sig, wake_c2.up},
	{PC.wake_c2.signal, wake_c2.up}	-> {PC.SPC.ok_sig, wake_c2.up},
};

// --- Queue state server ---
// Abstract queue: tracks presence of each type as binary flags.
// States: q_abc where a=count_A, b=count_B, c=count_C (each 0 or 1).
// Producers push their type, consumers pop according to their rules.
server:	queue (servers SPA:SPA, SPB:SPB, SPC:SPC, SC1:SC1, SC2:SC2; agents PA:PA, PB:PB, PC:PC, C1:C1, C2:C2),
services	{push_a, push_b, push_c, pop_a, pop_b, pop_c, pop_triple},
states	{q000, q100, q010, q001, q110, q101, q011, q111},
actions {
	// PA pushes type A: sets bit a from 0 to 1
	{PA.queue.push_a, queue.q000}		-> {PA.SPA.ok_push, queue.q100},
	{PA.queue.push_a, queue.q010}		-> {PA.SPA.ok_push, queue.q110},
	{PA.queue.push_a, queue.q001}		-> {PA.SPA.ok_push, queue.q101},
	{PA.queue.push_a, queue.q011}		-> {PA.SPA.ok_push, queue.q111},
	// PB pushes type B: sets bit b from 0 to 1
	{PB.queue.push_b, queue.q000}		-> {PB.SPB.ok_push, queue.q010},
	{PB.queue.push_b, queue.q100}		-> {PB.SPB.ok_push, queue.q110},
	{PB.queue.push_b, queue.q001}		-> {PB.SPB.ok_push, queue.q011},
	{PB.queue.push_b, queue.q101}		-> {PB.SPB.ok_push, queue.q111},
	// PC pushes type C: sets bit c from 0 to 1
	{PC.queue.push_c, queue.q000}		-> {PC.SPC.ok_push, queue.q001},
	{PC.queue.push_c, queue.q100}		-> {PC.SPC.ok_push, queue.q101},
	{PC.queue.push_c, queue.q010}		-> {PC.SPC.ok_push, queue.q011},
	{PC.queue.push_c, queue.q110}		-> {PC.SPC.ok_push, queue.q111},
	// C1 pops type A (when no C present — enforced by SC1 thread logic)
	{C1.queue.pop_a, queue.q100}		-> {C1.SC1.ok_pop, queue.q000},
	{C1.queue.pop_a, queue.q110}		-> {C1.SC1.ok_pop, queue.q010},
	{C1.queue.pop_a, queue.q101}		-> {C1.SC1.ok_pop, queue.q001},
	{C1.queue.pop_a, queue.q111}		-> {C1.SC1.ok_pop, queue.q011},
	// C1 pops type B (when no C present — enforced by SC1 thread logic)
	{C1.queue.pop_b, queue.q010}		-> {C1.SC1.ok_pop, queue.q000},
	{C1.queue.pop_b, queue.q110}		-> {C1.SC1.ok_pop, queue.q100},
	{C1.queue.pop_b, queue.q011}		-> {C1.SC1.ok_pop, queue.q001},
	{C1.queue.pop_b, queue.q111}		-> {C1.SC1.ok_pop, queue.q101},
	// C2 pops triple (A+B+C) — only from q111
	{C2.queue.pop_triple, queue.q111}	-> {C2.SC2.ok_pop3, queue.q000},
	// C2 pops single C
	{C2.queue.pop_c, queue.q001}		-> {C2.SC2.ok_pop1, queue.q000},
	{C2.queue.pop_c, queue.q011}		-> {C2.SC2.ok_pop1, queue.q010},
	{C2.queue.pop_c, queue.q101}		-> {C2.SC2.ok_pop1, queue.q100},
};

// --- Thread server: Producer A ---
// Protocol: wait(slots) -> wait(mutex) -> push_a -> signal(wake_c1) -> signal(mutex)
server:	SPA (servers mutex:mutex, slots:slots, wake_c1:wake_c1, queue:queue; agents PA:PA),
services	{start, ok_slots, ok_mutex, ok_push, ok_sig, ok_rel},
states	{ini, got_slots, in_cs, pushed, done},
actions {
	{PA.SPA.start, SPA.ini}			-> {PA.slots.wait, SPA.ini},
	{PA.SPA.ok_slots, SPA.ini}		-> {PA.mutex.wait, SPA.got_slots},
	{PA.SPA.ok_mutex, SPA.got_slots}	-> {PA.queue.push_a, SPA.in_cs},
	{PA.SPA.ok_push, SPA.in_cs}		-> {PA.wake_c1.signal, SPA.pushed},
	{PA.SPA.ok_sig, SPA.pushed}		-> {PA.mutex.signal, SPA.pushed},
	{PA.SPA.ok_rel, SPA.pushed}		-> {SPA.done},
};

// --- Thread server: Producer B ---
// Same protocol as PA, pushes type B, also signals wake_c1
server:	SPB (servers mutex:mutex, slots:slots, wake_c1:wake_c1, queue:queue; agents PB:PB),
services	{start, ok_slots, ok_mutex, ok_push, ok_sig, ok_rel},
states	{ini, got_slots, in_cs, pushed, done},
actions {
	{PB.SPB.start, SPB.ini}			-> {PB.slots.wait, SPB.ini},
	{PB.SPB.ok_slots, SPB.ini}		-> {PB.mutex.wait, SPB.got_slots},
	{PB.SPB.ok_mutex, SPB.got_slots}	-> {PB.queue.push_b, SPB.in_cs},
	{PB.SPB.ok_push, SPB.in_cs}		-> {PB.wake_c1.signal, SPB.pushed},
	{PB.SPB.ok_sig, SPB.pushed}		-> {PB.mutex.signal, SPB.pushed},
	{PB.SPB.ok_rel, SPB.pushed}		-> {SPB.done},
};

// --- Thread server: Producer C ---
// Same protocol, pushes type C, signals wake_c2 (not wake_c1)
server:	SPC (servers mutex:mutex, slots:slots, wake_c2:wake_c2, queue:queue; agents PC:PC),
services	{start, ok_slots, ok_mutex, ok_push, ok_sig, ok_rel},
states	{ini, got_slots, in_cs, pushed, done},
actions {
	{PC.SPC.start, SPC.ini}			-> {PC.slots.wait, SPC.ini},
	{PC.SPC.ok_slots, SPC.ini}		-> {PC.mutex.wait, SPC.got_slots},
	{PC.SPC.ok_mutex, SPC.got_slots}	-> {PC.queue.push_c, SPC.in_cs},
	{PC.SPC.ok_push, SPC.in_cs}		-> {PC.wake_c2.signal, SPC.pushed},
	{PC.SPC.ok_sig, SPC.pushed}		-> {PC.mutex.signal, SPC.pushed},
	{PC.SPC.ok_rel, SPC.pushed}		-> {SPC.done},
};

// --- Thread server: Consumer C1 ---
// Protocol: wait(wake_c1) -> wait(mutex) -> pop_a OR pop_b -> signal(slots) -> signal(mutex)
// C1 only consumes A or B elements (nondeterministic choice modeled by two actions from got_wc1)
server:	SC1 (servers mutex:mutex, slots:slots, wake_c1:wake_c1, queue:queue; agents C1:C1),
services	{start, ok_wc1, ok_mutex, ok_pop, ok_slots, ok_rel},
states	{ini, got_wc1, in_cs, popped, done},
actions {
	{C1.SC1.start, SC1.ini}			-> {C1.wake_c1.wait, SC1.ini},
	{C1.SC1.ok_wc1, SC1.ini}		-> {C1.mutex.wait, SC1.got_wc1},
	{C1.SC1.ok_mutex, SC1.got_wc1}		-> {C1.queue.pop_a, SC1.in_cs},
	{C1.SC1.ok_mutex, SC1.got_wc1}		-> {C1.queue.pop_b, SC1.in_cs},
	{C1.SC1.ok_pop, SC1.in_cs}		-> {C1.slots.signal, SC1.popped},
	{C1.SC1.ok_slots, SC1.popped}		-> {C1.mutex.signal, SC1.popped},
	{C1.SC1.ok_rel, SC1.popped}		-> {SC1.done},
};

// --- Thread server: Consumer C2 ---
// Protocol: wait(wake_c2) -> wait(mutex) -> pop_triple OR pop_c -> release slots -> signal(mutex)
// If pop_triple: releases 3 slots. If pop_c: releases 1 slot.
server:	SC2 (servers mutex:mutex, slots:slots, wake_c2:wake_c2, queue:queue; agents C2:C2),
services	{start, ok_wc2, ok_mutex, ok_pop3, ok_pop1, ok_slots, ok_rel},
states	{ini, got_wc2, in_cs, rel1, rel2, rel3, rel_single, done},
actions {
	{C2.SC2.start, SC2.ini}			-> {C2.wake_c2.wait, SC2.ini},
	{C2.SC2.ok_wc2, SC2.ini}		-> {C2.mutex.wait, SC2.got_wc2},
	{C2.SC2.ok_mutex, SC2.got_wc2}		-> {C2.queue.pop_triple, SC2.in_cs},
	{C2.SC2.ok_mutex, SC2.got_wc2}		-> {C2.queue.pop_c, SC2.in_cs},
	// Triple pop: release 3 slots one by one
	{C2.SC2.ok_pop3, SC2.in_cs}		-> {C2.slots.signal, SC2.rel1},
	{C2.SC2.ok_slots, SC2.rel1}		-> {C2.slots.signal, SC2.rel2},
	{C2.SC2.ok_slots, SC2.rel2}		-> {C2.slots.signal, SC2.rel3},
	{C2.SC2.ok_slots, SC2.rel3}		-> {C2.mutex.signal, SC2.rel3},
	// Single C pop: release 1 slot
	{C2.SC2.ok_pop1, SC2.in_cs}		-> {C2.slots.signal, SC2.rel_single},
	{C2.SC2.ok_slots, SC2.rel_single}	-> {C2.mutex.signal, SC2.rel_single},
	{C2.SC2.ok_rel, SC2.rel3}		-> {SC2.done},
	{C2.SC2.ok_rel, SC2.rel_single}	-> {SC2.done},
};

// --- Type and instance declarations ---
agent:	PA, PB, PC, C1, C2;
agents	PA:PA, PB:PB, PC:PC, C1:C1, C2:C2;
servers	mutex:mutex, slots:slots, wake_c1:wake_c1, wake_c2:wake_c2, queue:queue,
	SPA:SPA, SPB:SPB, SPC:SPC, SC1:SC1, SC2:SC2;

// --- Initial configuration ---
init	-> {
	PA.SPA.start,
	PB.SPB.start,
	PC.SPC.start,
	C1.SC1.start,
	C2.SC2.start,
	mutex(SPA,SPB,SPC,SC1,SC2,PA,PB,PC,C1,C2).up,
	slots(SPA,SPB,SPC,SC1,SC2,PA,PB,PC,C1,C2).s4,
	wake_c1(SPA,SPB,SC1,PA,PB,C1).down,
	wake_c2(SPC,SC2,PC,C2).down,
	queue(SPA,SPB,SPC,SC1,SC2,PA,PB,PC,C1,C2).q000,
	SPA(mutex,slots,wake_c1,queue,PA).ini,
	SPB(mutex,slots,wake_c1,queue,PB).ini,
	SPC(mutex,slots,wake_c2,queue,PC).ini,
	SC1(mutex,slots,wake_c1,queue,C1).ini,
	SC2(mutex,slots,wake_c2,queue,C2).ini,
}
