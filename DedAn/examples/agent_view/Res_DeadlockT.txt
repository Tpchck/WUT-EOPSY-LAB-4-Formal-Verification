services	{start, ok_take[2], used, do_work}
states	{ini, wait, using[2], own_work[2]}
;

actions {
	{A.proc.start, proc.ini}	-> {A.res.take, proc.wait},
	<j=1..2>{A.proc.do_work, proc.own_work[j]}	-> {A.proc.do_work, proc.own_work[j]},
	<j=1..2>{A.proc.do_work, proc.own_work[j]}	-> {A.res.use[j], proc.using[j]},

	<i=1..2>{A.res.use[i], res.taken[i]}	-> {A.res.use[i], res.taken_act[i]},
	<i=1..2>{A.res.take, res.taken_act[i]}	-> {A.proc.ok_take[3-i], res.taken1_2_act1_2},

	<i=1..2>{A.res.use[i], res.taken1_2}	-> {A.res.use[i], res.taken1_2_act[i]},
	<i=1..2>{A.res.use[i], res.taken1_2_act[3-i]}	-> {A.res.use[i], res.taken1_2_act1_2},
	<i=1..2>{A.res.use[i], res.taken1_2_act[i]}	-> {A.proc.used, res.taken1_2},
	<i=1..2>{A.res.use[i], res.taken1_2_act[i]}	-> {A.res.use[i], res.taken1_2_act[i]},
	<i=1..2>{A.res.use[i], res.taken1_2_act1_2}	-> {A.proc.used, res.taken1_2_act[3-i]},
};
agents	A[3]:A;

	A[1](proc[1],res).proc[1].start,
